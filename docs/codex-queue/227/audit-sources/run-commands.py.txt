runs=repo/'docs/codex-queue/227/runs';runs.mkdir(exist_ok=True)
env={**os.environ,'GIT_OPTIONAL_LOCKS':'0','pnpm_config_verify_deps_before_run':'false'}
groups={
 'node': [('portal-test',['pnpm','portal:test']),('learning-test',['pnpm','learning:test']),('learning-check',['pnpm','learning:check']),('test-216',['node','--test','scripts/validate-play-mode.test.mjs']),('play-mode',['pnpm','validate:play-mode']),('space-integration',['pnpm','validate:space-integration']),('game-dir-syntax',['bash','-n','scripts/game-dir.sh']),('bootstrap-syntax',['bash','-n','scripts/bootstrap.sh'])],

}
for item in groups[sys.argv[1]]:
 name,args,*where=item;cwd=repo/where[0] if where else repo;t=time.time()
 with (runs/(name+'.txt')).open('w') as out:
  out.write('cwd='+str(cwd)+'\nargv='+json.dumps(args,ensure_ascii=False)+'\nGIT_OPTIONAL_LOCKS=0 pnpm_config_verify_deps_before_run=false\n');out.flush()
  p=subprocess.run(args,cwd=cwd,env=env,stdout=out,stderr=subprocess.STDOUT)
  out.write('\nexit='+str(p.returncode)+'\n')
 text=(runs/(name+'.txt')).read_text()

 meta['counts']={k:int(v) for k,v in re.findall(r'^ℹ (tests|pass|fail|cancelled|skipped|todo) (\d+)',text,re.M)}
 (runs/(name+'.json')).write_text(json.dumps(meta,ensure_ascii=False,indent=2)+'\n')
