import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import {execFileSync} from 'node:child_process';
const repo=process.cwd();
const git=(args,cwd=repo)=>execFileSync('git',args,{cwd,env:{...process.env,GIT_OPTIONAL_LOCKS:'0'},encoding:'utf8'});
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
function info(p){
  try{
    const s=fs.lstatSync(p);
    if(s.isSymbolicLink()) return {kind:'symlink',target:fs.readlinkSync(p),sha256:sha(fs.readlinkSync(p))};
    if(s.isFile()) return {kind:'file',size:s.size,sha256:sha(fs.readFileSync(p))};
    if(s.isDirectory()){
      try{return {kind:'directory',head:git(['rev-parse','HEAD'],p).trim(),statusSha256:sha(git(['status','--porcelain'],p))};}
      catch{return {kind:'directory',gitUnavailable:true};}
    }
    return {kind:'other'};
  }catch(e){if(e.code==='ENOENT')return {kind:'deleted'};throw e;}
}
const raw=git(['status','--porcelain=v1','-z','-uall']);
const parts=raw.split('\0');const entries=[];
for(let i=0;i<parts.length;i++){
  if(!parts[i])continue;
  const status=parts[i].slice(0,2),p=parts[i].slice(3);
  const entry={status,path:p,...info(p)};
  if(/[RC]/.test(status)){entry.originalPath=parts[++i];}
  entries.push(entry);
}
function files(root){
  if(!fs.existsSync(root))return {exists:false,files:[]};
  const out=[];
  function walk(d){for(const n of fs.readdirSync(d).sort()){const p=path.join(d,n),s=fs.lstatSync(p);if(s.isDirectory())walk(p);else out.push({path:p,...info(p)});}}
  walk(root);return {exists:true,files:out};
}
const result={head:git(['rev-parse','HEAD']).trim(),branch:git(['branch','--show-current']).trim(),entries,portalDist:files('portal/dist'),viteTemp:files('portal/node_modules/.vite-temp')};
const out=process.argv[2];fs.mkdirSync(path.dirname(out),{recursive:true});fs.writeFileSync(out,JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify({out,head:result.head,branch:result.branch,entries:entries.length,portalDistFiles:result.portalDist.files.length,viteTempFiles:result.viteTemp.files.length}));
